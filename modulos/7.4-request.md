# 7.4 — Request Object: Parámetros, Headers y Validación

> **Objetivo**: Dominar el objeto Request de Magento — cómo leer parámetros GET/POST, headers, archivos subidos, validar inputs y manejar los diferentes tipos de request de forma segura.
> **Estado**: ⬜ Pendiente
> **Prerequisito**: [7.3 — ResultFactory: tipos de respuesta](7.3-result-factory.md)

---

## El RequestInterface

```php
use Magento\Framework\App\RequestInterface;
// o más específico:
use Magento\Framework\App\Request\Http;
```

🔄 **Laravel**: `$request->input('name')`, `$request->query('page')`, `$request->header('Accept')`.

Magento tiene una API similar pero más explícita sobre el origen del dato.

---

## Leer parámetros GET

```php
// URL: /blog/post/view?id=5&tab=comments&page=2

$id   = $this->request->getParam('id');           // "5" (string)
$id   = (int) $this->request->getParam('id');     // 5 (int — castear siempre)
$tab  = $this->request->getParam('tab', 'info');  // "comments" (con default)
$page = (int) $this->request->getParam('page', 1); // 2

// Todos los parámetros GET como array
$params = $this->request->getParams();
// ['id' => '5', 'tab' => 'comments', 'page' => '2']

// Parámetro de ruta (segmento de URL)
// URL: /blog/post/view/id/5  (formato Magento: key/value en el path)
$id = $this->request->getParam('id'); // también funciona con este formato
```

---

## Leer parámetros POST

```php
// POST body: title=Mi+Post&content=Contenido&status=draft

// Un campo específico
$title   = $this->request->getPost('title');
$content = $this->request->getPost('content');
$status  = $this->request->getPost('status', 'draft'); // con default

// Todo el POST body como array
$postData = $this->request->getPost()->toArray();
// ['title' => 'Mi Post', 'content' => 'Contenido', 'status' => 'draft']

// Alternativa — getParam() busca en GET y POST:
$title = $this->request->getParam('title'); // busca en ambos
```

## Leer body JSON (AJAX con Content-Type: application/json)

```php
// Request AJAX con body JSON:
// POST /blog/post/save
// Content-Type: application/json
// Body: {"title":"Mi Post","tags":["php","magento"]}

$body    = $this->request->getContent(); // string JSON crudo
$data    = json_decode($body, true);     // array PHP

$title   = $data['title'] ?? '';
$tags    = $data['tags'] ?? [];

// Validar que sea JSON válido:
if (json_last_error() !== JSON_ERROR_NONE) {
    // JSON inválido
}
```

## Leer headers

```php
// Leer un header específico
$accept        = $this->request->getHeader('Accept');
$contentType   = $this->request->getHeader('Content-Type');
$authorization = $this->request->getHeader('Authorization');
$customHeader  = $this->request->getHeader('X-My-Custom-Header');

// Verificar si es una request AJAX
$isAjax = $this->request->isAjax();
// true si el header X-Requested-With: XMLHttpRequest está presente

// Verificar el método HTTP
$method = $this->request->getMethod(); // "GET", "POST", "PUT", "DELETE"
$isPost = $this->request->isPost();
$isGet  = $this->request->isGet();

// Verificar si es HTTPS
$isSecure = $this->request->isSecure();
```

## Archivos subidos

```php
// Input: <input type="file" name="cover_image">

$files = $this->request->getFiles('cover_image');
// Equivalente a $_FILES['cover_image']

if ($files && $files['error'] === UPLOAD_ERR_OK) {
    $tmpPath  = $files['tmp_name'];
    $fileName = $files['name'];
    $fileSize = $files['size'];
    $mimeType = $files['type'];

    // Mover el archivo a pub/media/blog/
    $mediaPath = $this->filesystem
        ->getDirectoryRead(\Magento\Framework\App\Filesystem\DirectoryList::MEDIA)
        ->getAbsolutePath('blog/');

    move_uploaded_file($tmpPath, $mediaPath . $fileName);
}
```

## Validación de inputs

Magento no tiene un sistema de validación tan completo como Laravel, pero sí tiene helpers útiles:

```php
namespace MiVendor\Blog\Controller\Post;

use Magento\Framework\App\Action\HttpPostActionInterface;
use Magento\Framework\App\RequestInterface;
use Magento\Framework\Controller\Result\JsonFactory;
use Magento\Framework\Filter\StripTags;

class Save implements HttpPostActionInterface
{
    public function __construct(
        private readonly RequestInterface $request,
        private readonly JsonFactory      $jsonFactory,
        private readonly StripTags        $stripTags,
    ) {}

    public function execute(): \Magento\Framework\Controller\ResultInterface
    {
        $result = $this->jsonFactory->create();

        // 1. Leer y sanear inputs
        $title   = $this->stripTags->filter($this->request->getPost('title', ''));
        $content = $this->request->getPost('content', '');
        $status  = $this->request->getPost('status', 'draft');

        // 2. Validar manualmente
        $errors = $this->validate(['title' => $title, 'content' => $content, 'status' => $status]);

        if (!empty($errors)) {
            $result->setHttpResponseCode(422);
            return $result->setData(['success' => false, 'errors' => $errors]);
        }

        // 3. Procesar...
        return $result->setData(['success' => true]);
    }

    private function validate(array $data): array
    {
        $errors = [];

        if (empty(trim($data['title']))) {
            $errors['title'] = (string) __('El título es obligatorio.');
        } elseif (strlen($data['title']) < 3) {
            $errors['title'] = (string) __('El título debe tener al menos 3 caracteres.');
        } elseif (strlen($data['title']) > 255) {
            $errors['title'] = (string) __('El título no puede superar 255 caracteres.');
        }

        if (empty(trim($data['content']))) {
            $errors['content'] = (string) __('El contenido es obligatorio.');
        }

        $validStatuses = ['draft', 'published', 'archived'];
        if (!in_array($data['status'], $validStatuses, true)) {
            $errors['status'] = (string) __('Estado inválido.');
        }

        return $errors;
    }
}
```

## Sanitización — nunca confíes en el input

```php
use Magento\Framework\Filter\StripTags;
use Magento\Framework\Escaper;

class MiController implements HttpPostActionInterface
{
    public function __construct(
        private readonly RequestInterface $request,
        private readonly StripTags        $stripTags,
        private readonly Escaper          $escaper,
    ) {}

    public function execute(): \Magento\Framework\Controller\ResultInterface
    {
        // Eliminar tags HTML del input
        $title = $this->stripTags->filter($this->request->getPost('title'));

        // Escapar para output en HTML (en templates .phtml)
        // Nota: esto va en el template, no en el controller
        // echo $this->escaper->escapeHtml($title);

        // Castear tipos siempre
        $id     = (int) $this->request->getParam('id');        // nunca dejar como string
        $price  = (float) $this->request->getPost('price');
        $active = (bool) $this->request->getPost('is_active');
    }
}
```

## Acceder al request en otras clases (no solo Controllers)

```php
// Si necesitas el request fuera de un Controller (ej: en un Service):
// NO inyectes RequestInterface directamente en servicios — viola SRP
// En su lugar, pasa los datos como parámetros:

// ❌ Incorrecto — el servicio no debería conocer el request
class PostService
{
    public function __construct(private readonly RequestInterface $request) {}

    public function save(): void
    {
        $title = $this->request->getPost('title'); // acoplamiento al HTTP
    }
}

// ✅ Correcto — el Controller extrae los datos y los pasa al servicio
class PostService
{
    public function save(string $title, string $content): Post { ... }
}

// En el Controller:
$this->postService->save(
    $this->request->getPost('title'),
    $this->request->getPost('content')
);
```

## Request en el área Admin vs Frontend

```php
// Ambas áreas usan el mismo RequestInterface
// La diferencia está en cómo se valida el CSRF (Form Key)

// En el admin, Magento valida automáticamente el form key en POST
// En el frontend, debes manejarlo manualmente (ver módulo 7.5)

// Verificar si estás en el admin:
$isAdmin = $this->request->getRouteName() === 'adminhtml';

// Obtener el área actual:
// Inyecta \Magento\Framework\App\State en el constructor:
$area = $this->appState->getAreaCode(); // "frontend", "adminhtml", "webapi_rest"
```

## Resumen

```php
// GET params
$this->request->getParam('key', 'default');
$this->request->getParams();

// POST params
$this->request->getPost('key', 'default');
$this->request->getPost()->toArray();

// JSON body
json_decode($this->request->getContent(), true);

// Headers
$this->request->getHeader('Accept');
$this->request->isAjax();
$this->request->getMethod();

// Archivos
$this->request->getFiles('input_name');

// Seguridad:
// - Castear siempre: (int), (float), (bool)
// - Sanitizar: $stripTags->filter()
// - Nunca inyectar Request en servicios — pasar datos como parámetros
```

## Ejercicios

### Ejercicio 1 — Controller de búsqueda con validación

Crea un Controller `GET /blog/search/index` que:
1. Lee el parámetro `q` (query de búsqueda) y `page` (página, default 1)
2. Valida que `q` tenga al menos 3 caracteres
3. Si `q` es muy corto → redirige a `/blog` con un mensaje de error
4. Si es válido → retorna una `Page`

<details>
<summary>Ver solución</summary>

```php
<?php
// Controller/Search/Index.php
namespace MiVendor\Blog\Controller\Search;

use Magento\Framework\App\Action\HttpGetActionInterface;
use Magento\Framework\App\RequestInterface;
use Magento\Framework\View\Result\PageFactory;
use Magento\Framework\Controller\Result\RedirectFactory;
use Magento\Framework\Message\ManagerInterface;
use Magento\Framework\Filter\StripTags;

class Index implements HttpGetActionInterface
{
    public function __construct(
        private readonly RequestInterface $request,
        private readonly PageFactory      $pageFactory,
        private readonly RedirectFactory  $redirectFactory,
        private readonly ManagerInterface $messageManager,
        private readonly StripTags        $stripTags,
    ) {}

    public function execute(): \Magento\Framework\Controller\ResultInterface
    {
        // Leer y sanear parámetros
        $query = $this->stripTags->filter(
            trim($this->request->getParam('q', ''))
        );
        $page = max(1, (int) $this->request->getParam('page', 1));

        // Validar
        if (strlen($query) < 3) {
            $this->messageManager->addErrorMessage(
                __('La búsqueda debe tener al menos 3 caracteres.')
            );
            return $this->redirectFactory->create()->setPath('blog');
        }

        // Renderizar la página de resultados
        $resultPage = $this->pageFactory->create();
        $resultPage->getConfig()->getTitle()->set(
            __('Resultados para: %1', $query)
        );

        return $resultPage;
    }
}
```

</details>

### Ejercicio 2 — Endpoint AJAX con body JSON

Crea un Controller `POST /blog/post/update` que:
1. Lee el body como JSON (`{"id": 5, "title": "Nuevo título"}`)
2. Valida que `id` sea un entero positivo y `title` no esté vacío
3. Devuelve JSON con `{"success": true}` o `{"success": false, "errors": {...}}`

<details>
<summary>Ver solución</summary>

```php
<?php
// Controller/Post/Update.php
namespace MiVendor\Blog\Controller\Post;

use Magento\Framework\App\Action\HttpPostActionInterface;
use Magento\Framework\App\RequestInterface;
use Magento\Framework\Controller\Result\JsonFactory;
use Magento\Framework\Filter\StripTags;
use MiVendor\Blog\Api\PostRepositoryInterface;

class Update implements HttpPostActionInterface
{
    public function __construct(
        private readonly RequestInterface        $request,
        private readonly JsonFactory             $jsonFactory,
        private readonly StripTags               $stripTags,
        private readonly PostRepositoryInterface $postRepository,
    ) {}

    public function execute(): \Magento\Framework\Controller\ResultInterface
    {
        $result = $this->jsonFactory->create();

        // Leer y decodificar JSON body
        $body = $this->request->getContent();
        $data = json_decode($body, true);

        if (json_last_error() !== JSON_ERROR_NONE) {
            $result->setHttpResponseCode(400);
            return $result->setData(['success' => false, 'errors' => ['json' => 'Body JSON inválido']]);
        }

        // Extraer y validar
        $id    = (int) ($data['id'] ?? 0);
        $title = $this->stripTags->filter(trim($data['title'] ?? ''));

        $errors = [];
        if ($id <= 0) {
            $errors['id'] = 'ID debe ser un entero positivo';
        }
        if (empty($title)) {
            $errors['title'] = 'El título no puede estar vacío';
        }

        if (!empty($errors)) {
            $result->setHttpResponseCode(422);
            return $result->setData(['success' => false, 'errors' => $errors]);
        }

        // Procesar
        try {
            $post = $this->postRepository->getById($id);
            $post->setTitle($title);
            $this->postRepository->save($post);

            return $result->setData(['success' => true, 'id' => $id]);

        } catch (\Exception $e) {
            $result->setHttpResponseCode(404);
            return $result->setData(['success' => false, 'errors' => ['id' => 'Post no encontrado']]);
        }
    }
}
```

```bash
# Probar con curl:
curl -X POST http://localhost:8181/blog/post/update \
  -H "Content-Type: application/json" \
  -d '{"id": 5, "title": "Título actualizado"}'
# → {"success":true,"id":5}
```

</details>
