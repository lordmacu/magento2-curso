## El problema con las migrations clásicas

🔄 **Laravel**: Las migrations de Laravel son imperativos — describes lo que quieres hacer (`create_table`, `add_column`). Cada migration es una instrucción.

El sistema antiguo de Magento (`InstallSchema.php`, `UpgradeSchema.php`) tenía el mismo enfoque y creó problemas:
- Debías rastrear la versión manualmente en `module.xml`
- Si dos módulos modificaban la misma tabla, podían entrar en conflicto
- No había forma de "bajar" una migración

**Declarative Schema** (Magento 2.3+) cambia el enfoque:

```
Antes (imperativo):     "Agrega esta columna si no existe"
Ahora (declarativo):    "La tabla DEBE tener esta columna"

Magento calcula la diferencia y aplica el SQL necesario.
```

## Crear una tabla: estructura básica

```xml
<!-- app/code/MiVendor/Blog/etc/db_schema.xml -->
<?xml version="1.0"?>
<schema xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
        xsi:noNamespaceSchemaLocation="urn:magento:framework:Setup/Declaration/Schema/etc/schema.xsd">

    <table name="mivendor_blog_post" resource="default" engine="innodb"
           comment="Blog Posts Table">

        <!-- Primary Key -->
        <column xsi:type="int" name="post_id" padding="10" unsigned="true"
                nullable="false" identity="true" comment="Post ID"/>

        <!-- Strings -->
        <column xsi:type="varchar" name="title" nullable="false" length="255"
                comment="Post Title"/>
        <column xsi:type="varchar" name="url_key" nullable="false" length="255"
                comment="URL Key / Slug"/>
        <column xsi:type="text" name="content" nullable="true"
                comment="Post Content"/>
        <column xsi:type="text" name="short_description" nullable="true"
                comment="Short Description"/>

        <!-- Booleanos e Integers -->
        <column xsi:type="smallint" name="is_active" padding="5" unsigned="true"
                nullable="false" default="1" comment="Is Active"/>
        <column xsi:type="int" name="author_id" padding="10" unsigned="true"
                nullable="true" comment="Author Admin User ID"/>
        <column xsi:type="int" name="views_count" padding="10" unsigned="true"
                nullable="false" default="0" comment="Views Count"/>

        <!-- Decimales -->
        <column xsi:type="decimal" name="rating" scale="2" precision="4"
                nullable="true" comment="Average Rating (0.00-10.00)"/>

        <!-- Fechas -->
        <column xsi:type="timestamp" name="created_at" on_update="false"
                nullable="false" default="CURRENT_TIMESTAMP" comment="Created At"/>
        <column xsi:type="timestamp" name="updated_at" on_update="true"
                nullable="false" default="CURRENT_TIMESTAMP" comment="Updated At"/>
        <column xsi:type="date" name="published_at" nullable="true"
                comment="Publication Date"/>

        <!-- Primary Key constraint -->
        <constraint xsi:type="primary" referenceId="PRIMARY">
            <column name="post_id"/>
        </constraint>

        <!-- Índices -->
        <index referenceId="MIVENDOR_BLOG_POST_URL_KEY" indexType="btree">
            <column name="url_key"/>
        </index>
        <index referenceId="MIVENDOR_BLOG_POST_IS_ACTIVE" indexType="btree">
            <column name="is_active"/>
        </index>

        <!-- Unique constraint -->
        <constraint xsi:type="unique" referenceId="MIVENDOR_BLOG_POST_URL_KEY_UNIQUE">
            <column name="url_key"/>
        </constraint>

    </table>

</schema>
```

## Tipos de columna disponibles

```xml
<!-- Enteros -->
<column xsi:type="tinyint"  name="campo" padding="3" unsigned="true"/>  <!-- -128..127 / 0..255 -->
<column xsi:type="smallint" name="campo" padding="5" unsigned="true"/>  <!-- -32768..32767 / 0..65535 -->
<column xsi:type="int"      name="campo" padding="10" unsigned="true"/> <!-- ±2.1B / 0..4.2B -->
<column xsi:type="bigint"   name="campo" padding="20" unsigned="true"/> <!-- grandes IDs -->

<!-- Auto-increment (PK) -->
<column xsi:type="int" name="entity_id" padding="10" unsigned="true"
        nullable="false" identity="true"/>

<!-- Strings -->
<column xsi:type="varchar" name="campo" length="255" nullable="false" default=""/>
<column xsi:type="char"    name="campo" length="32"/>    <!-- longitud fija -->
<column xsi:type="text"    name="campo" nullable="true"/> <!-- hasta 64KB -->
<column xsi:type="mediumtext" name="campo"/>              <!-- hasta 16MB -->

<!-- Números con decimales -->
<column xsi:type="decimal" name="precio" scale="4" precision="12"/> <!-- precision total, scale = decimales -->
<column xsi:type="float"   name="campo"/>
<column xsi:type="real"    name="campo"/>

<!-- Fechas -->
<column xsi:type="date"      name="campo"/>                 <!-- YYYY-MM-DD -->
<column xsi:type="datetime"  name="campo"/>                 <!-- YYYY-MM-DD HH:MM:SS -->
<column xsi:type="timestamp" name="campo" on_update="false" default="CURRENT_TIMESTAMP"/>

<!-- Binarios / JSON -->
<column xsi:type="blob"      name="campo"/>
<column xsi:type="varbinary" name="campo" length="255"/>
```

## Claves foráneas

```xml
<table name="mivendor_blog_post" ...>
    <!-- ... columnas ... -->

    <!-- FK hacia admin_user (autor del post) -->
    <constraint xsi:type="foreign"
                referenceId="MIVENDOR_BLOG_POST_AUTHOR_ID_ADMIN_USER_USER_ID"
                table="mivendor_blog_post"
                column="author_id"
                referenceTable="admin_user"
                referenceColumn="user_id"
                onDelete="SET NULL"/>
</table>

<table name="mivendor_blog_post_category" ...>
    <!-- Tabla pivote post-categoría -->
    <column xsi:type="int" name="post_id" .../>
    <column xsi:type="int" name="category_id" .../>

    <!-- FK con CASCADE DELETE — al eliminar post, eliminar relaciones -->
    <constraint xsi:type="foreign"
                referenceId="MIVENDOR_BLOG_POST_CATEGORY_POST_ID"
                table="mivendor_blog_post_category"
                column="post_id"
                referenceTable="mivendor_blog_post"
                referenceColumn="post_id"
                onDelete="CASCADE"/>
</table>
```

**Opciones de `onDelete`**:
- `CASCADE` — eliminar registros hijos
- `SET NULL` — poner NULL en la FK
- `NO ACTION` / `RESTRICT` — error si hay registros hijos (default)

## Modificar una tabla existente (el poder declarativo)

No escribes una migration de "alterar" — simplemente **editas el XML**:

```xml
<!-- Antes — tabla original -->
<table name="mivendor_blog_post" ...>
    <column xsi:type="varchar" name="title" nullable="false" length="255"/>
    <!-- ... -->
</table>
```

```xml
<!-- Después — agregas featured_image y eliminas un índice, solo editas el XML -->
<table name="mivendor_blog_post" ...>
    <column xsi:type="varchar" name="title" nullable="false" length="255"/>
    <!-- Nueva columna — Magento hará el ALTER TABLE automáticamente -->
    <column xsi:type="varchar" name="featured_image" nullable="true" length="512"
            comment="Featured Image Path"/>
    <column xsi:type="int" name="comment_count" padding="10" unsigned="true"
            nullable="false" default="0" comment="Number of Comments"/>
    <!-- ... -->
</table>
```

```bash
# Aplicar los cambios
php bin/magento setup:upgrade
```

Magento compara el XML actual con el schema real en MySQL y genera el `ALTER TABLE` necesario.

## El archivo db_schema_whitelist.json

Para que los cambios destructivos (eliminar columnas/tablas) funcionen correctamente, necesitas el whitelist:

```bash
# Generar el whitelist — correr después de crear/modificar db_schema.xml
php bin/magento setup:db-declaration:generate-whitelist --module-name=MiVendor_Blog
```

Genera `etc/db_schema_whitelist.json`:

```json
{
    "mivendor_blog_post": {
        "column": {
            "post_id": true,
            "title": true,
            "url_key": true,
            "content": true,
            "is_active": true,
            "created_at": true,
            "updated_at": true
        },
        "constraint": {
            "PRIMARY": true,
            "MIVENDOR_BLOG_POST_URL_KEY_UNIQUE": true
        },
        "index": {
            "MIVENDOR_BLOG_POST_URL_KEY": true
        }
    }
}
```

**Sin el whitelist**: Magento no permite eliminar columnas (para evitar pérdida accidental de datos).

## Eliminar una columna o tabla

```xml
<!-- Para eliminar una columna: simplemente quitarla del XML
     Y tener el elemento en el whitelist.json -->

<!-- Para eliminar una tabla: quitar el bloque <table> del XML
     La tabla debe estar en el whitelist.json -->
```

```bash
# 1. Editar db_schema.xml (quitar la columna/tabla)
# 2. Regenerar el whitelist
php bin/magento setup:db-declaration:generate-whitelist --module-name=MiVendor_Blog
# 3. Aplicar
php bin/magento setup:upgrade
```

```xml

    <!-- ── Tabla de Posts ─────────────────────────────────── -->

           comment="Blog Posts">
        <column xsi:type="int"       name="post_id"      padding="10" unsigned="true" nullable="false" identity="true"/>
        <column xsi:type="int"       name="store_id"     padding="5"  unsigned="true" nullable="false" default="0"/>
        <column xsi:type="varchar"   name="title"        nullable="false" length="255"/>
        <column xsi:type="varchar"   name="url_key"      nullable="false" length="255"/>
        <column xsi:type="text"      name="content"      nullable="true"/>
        <column xsi:type="varchar"   name="meta_title"   nullable="true"  length="255"/>
        <column xsi:type="text"      name="meta_description" nullable="true"/>
        <column xsi:type="smallint"  name="is_active"    padding="5"  unsigned="true" nullable="false" default="1"/>
        <column xsi:type="int"       name="views_count"  padding="10" unsigned="true" nullable="false" default="0"/>
        <column xsi:type="timestamp" name="created_at"   nullable="false" default="CURRENT_TIMESTAMP" on_update="false"/>
        <column xsi:type="timestamp" name="updated_at"   nullable="false" default="CURRENT_TIMESTAMP" on_update="true"/>

        <constraint xsi:type="primary" referenceId="PRIMARY">
            <column name="post_id"/>
        </constraint>
        <constraint xsi:type="unique" referenceId="MIVENDOR_BLOG_POST_URL_KEY_STORE_ID">
            <column name="url_key"/>
            <column name="store_id"/>
        </constraint>
        <index referenceId="MIVENDOR_BLOG_POST_IS_ACTIVE" indexType="btree">
            <column name="is_active"/>
        </index>
        <index referenceId="MIVENDOR_BLOG_POST_CREATED_AT" indexType="btree">
            <column name="created_at"/>
        </index>
    </table>

    <!-- ── Tabla de Categorías ────────────────────────────── -->
    <table name="mivendor_blog_category" resource="default" engine="innodb"
           comment="Blog Categories">
        <column xsi:type="int"     name="category_id" padding="10" unsigned="true" nullable="false" identity="true"/>
        <column xsi:type="varchar" name="name"         nullable="false" length="255"/>
        <column xsi:type="varchar" name="url_key"      nullable="false" length="255"/>
        <column xsi:type="smallint" name="is_active"   padding="5" unsigned="true" nullable="false" default="1"/>
        <column xsi:type="int"     name="sort_order"   padding="10" unsigned="true" nullable="false" default="0"/>

        <constraint xsi:type="primary" referenceId="PRIMARY">
            <column name="category_id"/>
        </constraint>
    </table>

    <!-- ── Tabla pivote Post-Categoría ───────────────────── -->
    <table name="mivendor_blog_post_category" resource="default" engine="innodb"
           comment="Post to Category Relations">
        <column xsi:type="int" name="post_id"     padding="10" unsigned="true" nullable="false"/>
        <column xsi:type="int" name="category_id" padding="10" unsigned="true" nullable="false"/>

        <constraint xsi:type="primary" referenceId="PRIMARY">
            <column name="post_id"/>
            <column name="category_id"/>
        </constraint>
        <constraint xsi:type="foreign"
                    referenceId="MIVENDOR_BLOG_POST_CATEGORY_POST_ID"
                    table="mivendor_blog_post_category" column="post_id"
                    referenceTable="mivendor_blog_post" referenceColumn="post_id"
                    onDelete="CASCADE"/>
        <constraint xsi:type="foreign"
                    referenceId="MIVENDOR_BLOG_POST_CATEGORY_CATEGORY_ID"
                    table="mivendor_blog_post_category" column="category_id"
                    referenceTable="mivendor_blog_category" referenceColumn="category_id"
                    onDelete="CASCADE"/>
```

## Resumen

```bash
# Flujo completo de trabajo con Declarative Schema:

# 1. Crear/editar etc/db_schema.xml
# 2. Generar whitelist (necesario para cambios destructivos)
php bin/magento setup:db-declaration:generate-whitelist --module-name=MiVendor_Blog

# 3. Simular (opcional)
php bin/magento setup:upgrade --dry-run

# 4. Aplicar
php bin/magento setup:upgrade
```

```
Añadir columna     →  agregar <column> al XML
Modificar columna  →  editar atributos del <column>
Eliminar columna   →  quitar del XML + regenerar whitelist
Añadir índice      →  agregar <index> al XML
Añadir FK          →  agregar <constraint xsi:type="foreign">
```

### Ejercicio 1 — Schema para un módulo de FAQs

Crea el `db_schema.xml` para un módulo de preguntas frecuentes con:
- Tabla `mivendor_faq_question`: id, title (255), content (text), is_active, sort_order (int), created_at
- Tabla `mivendor_faq_category`: id, name (255), is_active, sort_order
- Tabla pivote `mivendor_faq_question_category` con FK y CASCADE

<details>
<summary>Ver solución</summary>

```xml
<?xml version="1.0"?>
<schema xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
        xsi:noNamespaceSchemaLocation="urn:magento:framework:Setup/Declaration/Schema/etc/schema.xsd">

    <table name="mivendor_faq_question" resource="default" engine="innodb"
           comment="FAQ Questions">
        <column xsi:type="int"       name="question_id" padding="10" unsigned="true" nullable="false" identity="true"/>
        <column xsi:type="varchar"   name="title"       nullable="false" length="255" comment="Question Title"/>
        <column xsi:type="text"      name="content"     nullable="true" comment="Answer Content"/>
        <column xsi:type="smallint"  name="is_active"   padding="5" unsigned="true" nullable="false" default="1"/>
        <column xsi:type="int"       name="sort_order"  padding="10" unsigned="true" nullable="false" default="0"/>
        <column xsi:type="timestamp" name="created_at"  nullable="false" default="CURRENT_TIMESTAMP" on_update="false"/>

        <constraint xsi:type="primary" referenceId="PRIMARY">
            <column name="question_id"/>
        </constraint>
        <index referenceId="MIVENDOR_FAQ_QUESTION_IS_ACTIVE" indexType="btree">
            <column name="is_active"/>
        </index>
        <index referenceId="MIVENDOR_FAQ_QUESTION_SORT_ORDER" indexType="btree">
            <column name="sort_order"/>
        </index>

        <column xsi:type="varchar"  name="name"        nullable="false" length="255" comment="Category Name"/>
        <column xsi:type="smallint" name="is_active"   padding="5" unsigned="true" nullable="false" default="1"/>
        <column xsi:type="int"      name="sort_order"  padding="10" unsigned="true" nullable="false" default="0"/>
```

### Ejercicio 2 — Agregar columnas a una tabla existente

El módulo Blog ya está en producción con su tabla `mivendor_blog_post`. Necesitas agregar:
- `featured_image` (varchar 512, nullable)
- `comment_count` (int unsigned, default 0)
- Un índice en `featured_image` para queries rápidas

¿Qué cambias y qué comandos corres?

<details>
<summary>Ver solución</summary>

```xml
<!-- etc/db_schema.xml — solo agregar las nuevas columnas al bloque existente -->
<table name="mivendor_blog_post" ...>
    <!-- ... columnas existentes sin modificar ... -->

    <!-- Nuevas columnas -->
    <column xsi:type="varchar" name="featured_image" nullable="true" length="512"
            comment="Featured Image Path"/>
    <column xsi:type="int" name="comment_count" padding="10" unsigned="true"
            nullable="false" default="0" comment="Number of Comments"/>

    <!-- Nuevo índice -->
    <index referenceId="MIVENDOR_BLOG_POST_FEATURED_IMAGE" indexType="btree">
        <column name="featured_image"/>
    </index>
</table>
```

```bash
# Regenerar whitelist para incluir las nuevas columnas e índice
php bin/magento setup:db-declaration:generate-whitelist --module-name=MiVendor_Blog

# Ver qué SQL se va a ejecutar
php bin/magento setup:upgrade --dry-run
# → ALTER TABLE `mivendor_blog_post` ADD COLUMN `featured_image` VARCHAR(512) ...
# → ALTER TABLE `mivendor_blog_post` ADD COLUMN `comment_count` INT UNSIGNED NOT NULL DEFAULT 0 ...
# → CREATE INDEX `MIVENDOR_BLOG_POST_FEATURED_IMAGE` ON `mivendor_blog_post` (`featured_image`)

# Aplicar
php bin/magento setup:upgrade
```

**Lo que NO haces**: crear un `UpgradeSchema.php` ni cambiar el `setup_version` en `module.xml` — el sistema declarativo lo maneja solo.

</details>
